TORLES ES MODOSITAS LISTABAN                                    [csak ZH2]
=============================================================================
Termek: nev, kategoria (string); ar (int); akcios (bool); datum (DateTime)

EGY ELEM TORLESE NEV SZERINT (kis/nagybetu-fuggetlenul)
    static bool Torol(List<Termek> lista, string nev)
    {
        for (int i = 0; i < lista.Count; i++)
        {
            if (lista[i].nev.ToLower() == nev.Trim().ToLower())
            {
                lista.RemoveAt(i);
                return true;
            }
        }
        return false;
    }

HIVAS FELHASZNALOI BEKERESSEL
    List<Termek> lista = new List<Termek>();
    Console.Write("Torlendo termek neve: ");
    string mit = Console.ReadLine();
    if (Torol(lista, mit))
        Console.WriteLine("Torolve.");
    else
        Console.WriteLine("Nincs ilyen termek.");

TOBB ELEM TORLESE: HATULROL ELORE (elore haladva kihagyna elemeket)
    List<Termek> lista = new List<Termek>();
    for (int i = lista.Count - 1; i >= 0; i--)
        if (lista[i].ar > 1000)
            lista.RemoveAt(i);
