  using System.IO;   (datumhoz: using System.Globalization;)

  TELJES BEOLVASAS (fejleces CSV, ; elvalaszto)
  Termek: nev, kategoria (string); ar (int); akcios (bool); datum (DateTime)
  Sor a fajlban:  Kifli;pekaru;45;igen;2024.12.17
    static List<Termek> Beolvas(string fajlnev)
    {
        List<Termek> lista = new List<Termek>();
        StreamReader sr = new StreamReader(fajlnev);
        sr.ReadLine();                              // fejlec eldobasa
        while (!sr.EndOfStream)
        {
            string[] t = sr.ReadLine().Split(';');
            Termek x = new Termek();
            x.nev = t[0];
            x.kategoria = t[1];
            x.ar = int.Parse(t[2]);
            x.akcios = t[3] == "igen";
            x.datum = DateTime.ParseExact(t[4], "yyyy.MM.dd",
                                          CultureInfo.InvariantCulture);
            lista.Add(x);
        }
        sr.Close();
        return lista;
    }
  Main-ben:   List<Termek> lista = Beolvas("termekek.csv");
