=============================================================================
Termek: nev, kategoria (string); ar (int); akcios (bool); datum (DateTime)

LISTA (ZH2): kategoriak, mindegyik egyszer
    static List<string> Kategoriak(List<Termek> lista)
    {
        List<string> egyedi = new List<string>();
        foreach (Termek x in lista)
        {
            bool marVan = false;
            for (int i = 0; i < egyedi.Count && !marVan; i++)
                if (egyedi[i] == x.kategoria)
                    marVan = true;
            if (!marVan)
                egyedi.Add(x.kategoria);
        }
        return egyedi;
    }
  (A belso for helyett: if (!egyedi.Contains(x.kategoria)) - csak ZH2-ben,
  es csak ha szabad.)

RENDEZVE / FORDITOTT SORRENDBEN / DARABSZAMMAL
    List<Termek> lista = new List<Termek>();
    List<string> kat = Kategoriak(lista);
    kat.Sort();                       // abc szerint
    kat.Reverse();                    // forditott (Sort utan: Z -> A)
    foreach (string k in kat)
    {
        int db = 0;
        foreach (Termek x in lista)
            if (x.kategoria == k)
                db++;
        Console.WriteLine($"{k}: {db} db");
    }

TOMB (ZH1): egyedi szamok egy masik tombbe
    int[] t = { 4, -2, 7, 7, 1, 4 };
    int[] egyedi = new int[t.Length];      // legfeljebb ennyi lehet
    int db = 0;                            // ennyi van tenylegesen
    for (int i = 0; i < t.Length; i++)
    {
        bool marVan = false;
        for (int j = 0; j < db; j++)
            if (egyedi[j] == t[i])
                marVan = true;
        if (!marVan)
        {
            egyedi[db] = t[i];
            db++;
        }
    }
    for (int i = 0; i < db; i++)           // NEM egyedi.Length-ig!
        Console.Write(egyedi[i] + " ");    // 4 -2 7 1
