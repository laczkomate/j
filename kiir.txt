  Interpolacio ($ jel): {kifejezes} behelyettesites
    int x = 7;
    double ar = 1234.5678;
    string nev = "Ede";
    Console.WriteLine($"Szia {nev}! x={x}, ketszerese={x * 2}");
    Console.WriteLine($"Ar: {ar:F2} Ft");          // 2 tizedes: 1234,57
    Console.WriteLine($"Ar: {ar:F0} Ft");          // 0 tizedes (kerekit)
    Console.WriteLine($"{x:D3}");                  // 007 (egesz, min. 3 jegy)
    Console.WriteLine($"[{nev,8}] [{nev,-8}]");    // jobbra / balra igazitas
    Console.WriteLine("x={0}, y={1}", x, ar);   // regi forma (orai)
