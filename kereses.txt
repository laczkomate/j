TOMB (ZH1)
    int[] t = { 4, -2, 7, 7, 1 };
    // VAN-E negativ? (eldontes)
    bool van = false;
    for (int i = 0; i < t.Length && !van; i++)
        if (t[i] < 0)
            van = true;
    // MIND pozitiv? (= nincs ellenpelda)
    bool mind = true;
    for (int i = 0; i < t.Length && mind; i++)
        if (t[i] <= 0)
            mind = false;
    // HOL az elso 7-es? (linearis kereses, -1 ha nincs)
    int idx = -1;
    int j = 0;
    while (j < t.Length && idx == -1)
    {
        if (t[j] == 7)
            idx = j;
        j++;
    }
    if (idx != -1)
        Console.WriteLine($"Az elso 7-es indexe: {idx}");
    else
        Console.WriteLine("Nincs 7-es.");

KIVALASZTAS (biztosan van ilyen elem, nem kell hatar-ellenorzes)
    int[] t = { 4, -2, 7, 7, 1 };

        k++;

    static int KeresIdx(List<Termek> lista, string nev)
    {
        for (int i = 0; i < lista.Count; i++)
            if (lista[i].nev.ToLower() == nev.Trim().ToLower())
                return i;
        return -1;
    }

    static bool Megfelel(Termek x, int maxAr)
    {
        return x.akcios && x.ar <= maxAr && x.kategoria != "pekaru";
    }
    static void Kereso(List<Termek> lista)
    {
        Console.Write("Legfeljebb mennyi lehet az ar? ");
        int maxAr = int.Parse(Console.ReadLine());
        int talalat = 0;
        foreach (Termek x in lista)
        {
            if (Megfelel(x, maxAr))
            {
                Console.WriteLine($"{x.nev} - {x.ar} Ft");
                talalat++;
            }
        }
        if (talalat == 0)
            Console.WriteLine("Nincs megfelelo termek.");
    }
