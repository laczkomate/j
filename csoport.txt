TOMB, SORFOLYTONOS (ZH1): CSOP csoport, mindegyikben MERET elem
  index = cs * MERET + j    (cs: csoport 0-tol, j: elem a csoporton belul)
    const int CSOP = 3;
    const int MERET = 4;
    Random rnd = new Random();
    int[] t = new int[CSOP * MERET];
    for (int i = 0; i < t.Length; i++)
        t[i] = rnd.Next(1, 101);              // 1..100
    int legjobbCs = 0;
    int legjobbOssz = 0;
    for (int cs = 0; cs < CSOP; cs++)
    {
        int ossz = 0;                         // csoportonkent nullazva!
        int max = t[cs * MERET];
        for (int j = 0; j < MERET; j++)
        {
            int ertek = t[cs * MERET + j];
            Console.Write($"{ertek,4}");
            ossz += ertek;
            if (ertek > max)
                max = ertek;
        }
        double atlag = (double)ossz / MERET;
        Console.WriteLine($" | ossz: {ossz}, atlag: {atlag:F2}, max: {max}");
        if (cs == 0 || ossz > legjobbOssz)
        {
            legjobbOssz = ossz;
            legjobbCs = cs;
        }
    }
    Console.WriteLine($"Legnagyobb osszeg: {legjobbCs + 1}. csoport");

LISTA (ZH2): kategoriankent a legolcsobb termek
    static void LegolcsobbKategoriankent(List<Termek> lista)
    {
        List<string> kat = new List<string>();     // egyedi kategoriak
        foreach (Termek x in lista)
        {
            bool marVan = false;
            foreach (string k in kat)
                if (k == x.kategoria)
                    marVan = true;
            if (!marVan)
                kat.Add(x.kategoria);
        }
        foreach (string k in kat)                  // csoportonkent minimum
        {
            int idx = -1;
            for (int i = 0; i < lista.Count; i++)
                if (lista[i].kategoria == k)
                    if (idx == -1 || lista[i].ar < lista[idx].ar)
                        idx = i;
            Console.WriteLine($"{k}: {lista[idx].nev} ({lista[idx].ar} Ft)");
        }
    }
