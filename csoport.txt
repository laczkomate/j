    const int CSOP = 3;
    const int MERET = 4;
    Random rnd = new Random();
    int[] t = new int[CSOP * MERET];
    for (int i = 0; i < t.Length; i++)
        t[i] = rnd.Next(1, 101);              // 1..100
    int legjobbCs = 0;
    int legjobbOssz = 0;
    for (int cs = 0; cs < CSOP; cs++)
    {
        int ossz = 0;                         // csoportonkent nullazva!
        int max = t[cs * MERET];
        for (int j = 0; j < MERET; j++)
        {
            int ertek = t[cs * MERET + j];
            Console.Write($"{ertek,4}");
            ossz += ertek;
            if (ertek > max)
                max = ertek;
        }
        double atlag = (double)ossz / MERET;
        Console.WriteLine($" | ossz: {ossz}, atlag: {atlag:F2}, max: {max}");
        if (cs == 0 || ossz > legjobbOssz)
        {
            legjobbOssz = ossz;
            legjobbCs = cs;
        }
    }
    Console.WriteLine($"Legnagyobb osszeg: {legjobbCs + 1}. csoport");

K VELETLEN ELEM NOVELESE CSOPORTONKENT, KORLATTAL, ELOTTE-UTANA ATLAG
    const int CSOP = 3;
    const int MERET = 30;
    Random rnd = new Random();
    double[] t = new double[CSOP * MERET];
    for (int i = 0; i < t.Length; i++)
        t[i] = rnd.Next(500, 1001) / 10.0;
    int javult = 0;
    for (int cs = 0; cs < CSOP; cs++)
    {
        double elotte = 0;
        for (int j = 0; j < MERET; j++)
            elotte += t[cs * MERET + j];
        elotte /= MERET;
        for (int k = 0; k < 5; k++)                // 5 veletlen elem
        {
            int idx = cs * MERET + rnd.Next(MERET);
            t[idx] += rnd.Next(1, 9);              // +1..8
            if (t[idx] > 100)
                t[idx] = 100;                      // csokkentesnel: < 0 -> 0
        }
        double utana = 0;
        for (int j = 0; j < MERET; j++)
            utana += t[cs * MERET + j];
        utana /= MERET;
        if (utana - elotte >= 2)
            javult++;
    }
    Console.WriteLine($"{javult} csoportnal javult legalabb 2-vel");

HANY CSOPORT / VAN-E CSOPORT, AHOL MIND ... (pl. mindegyik 80 felett)
    const int MERET = 4;
    int[] t = { 85, 90, 81, 99, 50, 95, 88, 91 };
    int csopDb = t.Length / MERET;
    int joCsop = 0;
    for (int cs = 0; cs < csopDb; cs++)
    {
        bool mind = true;
        for (int j = 0; j < MERET && mind; j++)
            if (t[cs * MERET + j] <= 80)
                mind = false;
        if (mind)
            joCsop++;
    }
    if (joCsop > 0)
        Console.WriteLine($"Van ilyen csoport ({joCsop} db).");
    else
        Console.WriteLine("Nincs ilyen csoport.");
