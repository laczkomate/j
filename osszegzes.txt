OSSZEGZES, MEGSZAMOLAS, ATLAG, SZAZALEK                             [ZH1+2]
=============================================================================
Termek: nev, kategoria (string); ar (int); akcios (bool); datum (DateTime)

TOMB (ZH1): osszeg, darab feltetellel, atlag, szazalek
    int[] t = { 4, -2, 7, 7, 1 };
    int ossz = 0;
    int db = 0;
    for (int i = 0; i < t.Length; i++)
    {
        ossz += t[i];                         // osszegzes
        if (t[i] > 3)                         // megszamolas
            db++;
    }
    double atlag = (double)ossz / t.Length;   // (double) nelkul csonkol!
    double szazalek = 100.0 * db / t.Length;
    Console.WriteLine($"Osszeg: {ossz}, atlag: {atlag:F2}");
    Console.WriteLine($"3-nal nagyobb: {db} db ({szazalek:F1}%)");

FELTETELES ATLAG (ugyanaz a feltetel az osszegnel es a darabnal)
    int[] t = { 4, -2, 7, 7, 1 };
    int ossz = 0;
    int db = 0;
    for (int i = 0; i < t.Length; i++)
    {
        if (t[i] > 0)
        {
            ossz += t[i];
            db++;
        }
    }
    if (db > 0)
        Console.WriteLine($"Pozitivak atlaga: {(double)ossz / db:F2}");
    else
        Console.WriteLine("Nincs pozitiv elem.");

LISTA + OBJEKTUM (ZH2): kategoria atlagara, fuggvenykent
    static double AtlagAr(List<Termek> lista, string kategoria)
    {
        int ossz = 0;
        int db = 0;
        foreach (Termek x in lista)
        {
            if (x.kategoria == kategoria)
            {
                ossz += x.ar;
                db++;
            }
        }
        if (db == 0)
            return 0;                     // nincs talalat: ne osszon 0-val
        return (double)ossz / db;
    }

IDOSZAKRA SZURT ATLAG (datum ket hatar kozott)
    List<Termek> lista = new List<Termek>();
    DateTime tol = new DateTime(2024, 1, 1);
    DateTime ig = new DateTime(2024, 6, 30);
    int ossz = 0;
    int db = 0;
    foreach (Termek x in lista)
    {
        if (x.datum >= tol && x.datum <= ig)
        {
            ossz += x.ar;
            db++;
        }
    }
    if (db > 0)
        Console.WriteLine($"Atlagar: {(double)ossz / db:F2} Ft");
    else
        Console.WriteLine("Nincs adat az idoszakban.");
