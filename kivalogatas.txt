    static List<Termek> Akciosak(List<Termek> lista)
    {
        List<Termek> eredmeny = new List<Termek>();
        foreach (Termek x in lista)
            if (x.akcios)
                eredmeny.Add(x);
        return eredmeny;
    }

    static void Valogat(List<Termek> forras, List<Termek> cel, int maxAr)
    {
        cel.Clear();                         // ELOSZOR uriteni!
        foreach (Termek x in forras)
            if (x.ar <= maxAr)
                cel.Add(x);
    }
