    bool van = false;
    int min = 0;
    int minIdx = -1;
    for (int i = 0; i < t.Length; i++)
    {
        if (t[i] % 2 == 0)
        {
            if (!van || t[i] < min)
            {
                min = t[i];
                minIdx = i;
                van = true;
            }
        }
    }
    if (van)
        Console.WriteLine($"Legkisebb paros: {min} (index: {minIdx})");
    else
        Console.WriteLine("Nincs paros szam.");

LISTA + OBJEKTUM (ZH2): legdragabb egy kategoriaban, indexszel
    static int LegdragabbIdx(List<Termek> lista, string kategoria)
    {
        int idx = -1;
        for (int i = 0; i < lista.Count; i++)
        {
            if (lista[i].kategoria == kategoria)
            {
                if (idx == -1 || lista[i].ar > lista[idx].ar)
                    idx = i;
            }
        }
        return idx;                 // -1: nincs ilyen kategoria
    }

HASZNALAT + KET CSOPORT SZELSOERTEKENEK OSSZEHASONLITASA
    List<Termek> lista = new List<Termek>();
    int a = LegdragabbIdx(lista, "tejtermek");
    int b = LegdragabbIdx(lista, "pekaru");
    if (a == -1 || b == -1)
        Console.WriteLine("Valamelyik kategoria ures.");
    else if (lista[a].ar > lista[b].ar)
        Console.WriteLine($"Dragabb: {lista[a].nev} ({lista[a].ar} Ft)");
    else if (lista[a].ar < lista[b].ar)
        Console.WriteLine($"Dragabb: {lista[b].nev} ({lista[b].ar} Ft)");
    else
        Console.WriteLine("Egyforma a legmagasabb ar.");
