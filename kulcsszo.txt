STOP-IG VAGY KAPACITASIG (do-while)
    const int KAPACITAS = 10;
    int db = 0;
    int ossz = 0;
    string be;
    do
    {
        Console.Write("Ertek (STOP = vege): ");
        be = Console.ReadLine().Trim();
        if (be.ToUpper() != "STOP")
        {
            ossz += int.Parse(be);
            db++;
        }
    } while (be.ToUpper() != "STOP" && db < KAPACITAS);
    if (db == KAPACITAS)
        Console.WriteLine("Megtelt.");
    Console.WriteLine($"{db} db, osszesen {ossz}");

VEGTELEN CIKLUS + break (ha tobb kulcsszo van)
    int db = 0;
    while (true)
    {
        Console.Write("Nev (vegeztem = kilep): ");
        string s = Console.ReadLine().Trim().ToLower();
        if (s == "vegeztem")
            break;
        db++;
    }

EGYSEG-VALTAS: "kovetkezo" kulcsszora VAGY ha nem fer el (doboz 1000 g)
    const int DOBOZ = 1000;
    int doboz = 1;
    int toltes = 0;
    while (true)
    {
        Console.Write("Suly (g) / kovetkezo / vege: ");
        string s = Console.ReadLine().Trim().ToLower();
        if (s == "vege")
            break;
        if (s == "kovetkezo")
        {
            doboz++;
            toltes = 0;
            continue;
        }
        int g = int.Parse(s);
        if (toltes + g > DOBOZ)              // nem fer el: uj doboz
        {
            doboz++;
            toltes = 0;
        }
        toltes += g;
    }
    Console.WriteLine($"Dobozok: {doboz}, szabad: {DOBOZ - toltes} g");
