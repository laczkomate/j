STOP-IG VAGY KAPACITASIG (do-while)
    const int KAPACITAS = 10;
    int db = 0;
    int ossz = 0;
    string be;
    do
    {
        Console.Write("Ertek (STOP = vege): ");
        be = Console.ReadLine().Trim();
        if (be.ToUpper() != "STOP")
        {
            ossz += int.Parse(be);
            db++;
        }
    } while (be.ToUpper() != "STOP" && db < KAPACITAS);
    if (db == KAPACITAS)
        Console.WriteLine("Megtelt.");
    Console.WriteLine($"{db} db, osszesen {ossz}");

EGYSEG-VALTAS: "kovetkezo" kulcsszora VAGY ha nem fer el (doboz 1000 g)
    const int DOBOZ = 1000;
    int doboz = 1;
    int toltes = 0;
    while (true)
    {
        Console.Write("Suly (g) / kovetkezo / vege: ");
        string s = Console.ReadLine().Trim().ToLower();
        if (s == "vege")
            break;
        if (s == "kovetkezo")
        {
            doboz++;
            toltes = 0;
            continue;
        }
        int g = int.Parse(s);
        if (toltes + g > DOBOZ)              // nem fer el: uj doboz
        {
            doboz++;
            toltes = 0;
        }
        toltes += g;
    }
    Console.WriteLine($"Dobozok: {doboz}, szabad: {DOBOZ - toltes} g");

KAPACITAS KET FELTETELLEL + SZABAD HELY KIIRASA MINDEN BEKERES ELOTT
    const int MAXDB = 10;
    const int MAXWH = 1000;
    int db = 0, wh = 0, resz = 1, osszDb = 0, osszWh = 0;
    while (true)
    {
        Console.WriteLine($"{resz}. resz: szabad {MAXDB - db} db, " +
                          $"{MAXWH - wh} Wh");
        Console.Write("Szenzor fogyasztasa (Wh) / uj / STOP: ");
        string s = Console.ReadLine().Trim();
        if (s.ToUpper() == "STOP")
            break;
        if (s.ToLower() == "uj")
        {
            resz++; db = 0; wh = 0;
            continue;
        }
        int e = int.Parse(s);
        if (db == MAXDB || wh + e > MAXWH)       // nem fer el: uj resz
        {
            resz++; db = 0; wh = 0;
        }
        db++; wh += e; osszDb++; osszWh += e;
    }
    Console.WriteLine($"{osszDb} szenzor, {resz} reszben, {osszWh} Wh");

CSAK MEGENGEDETT ERTEKEK TOMBBOL + FIZETES TULFIZETES NELKUL
    int[] ermek = { 10, 20, 50, 100, 200 };
    int hatra = 370;
    while (hatra > 0)
    {
        Console.Write($"Meg fizetendo: {hatra}. Penz: ");
        int e;
        bool szam = int.TryParse(Console.ReadLine(), out e);
        bool ervenyes = false;
        for (int i = 0; i < ermek.Length; i++)
            if (szam && ermek[i] == e)
                ervenyes = true;
        if (!ervenyes)
            Console.WriteLine("Nincs ilyen cimlet!");
        else if (e > hatra)
            Console.WriteLine("Tulfizetes, nem fogadom el!");
        else
            hatra -= e;
    }
    Console.WriteLine("Az osszeg kifizetve.");
