SZAZALEK (telitettseg), TELJES ADAG, ATLAG EGY ELEMRE, MERTEKEGYSEG
    static int F2_Telitettseg(Aru a)
    {
        return (int)Math.Round(100.0 * a.keszlet / a.maxKeszlet);
    }
    static int F2_TeljesEnergia(Aru a)          // a kcal 100 g-ra van megadva
    {
        return a.tomeg * a.kcal100 / 100;
    }
    static double F2_ElemAr(Aru a)              // egesz kell? a.ar / a.elemek
    {
        return (double)a.ar / a.elemek;
    }
    static double Kg(int gramm)
    {
        return gramm / 1000.0;                  // 1000.0! (int/int csonkol)
    }

KEDVEZMENYEK, EGYMASTOL FUGGETLEN FELTETELEK (mind teljesulhet)
    static int F3_KedvezmenyesAr(Aru a)
    {
        double ar = a.ar;
        if (a.lejarat < new DateTime(2022, 10, 1))
            ar = ar * 0.75;                       // -25%
        if (a.tipus == "HDD" || a.tipus == "SSHD")
            ar = ar - 12000;                      // fix osszeg
        if (a.kiallitasi)
            ar = ar * 0.85;                       // -15%
        return (int)Math.Round(ar);               // nincs kedvezmeny: eredeti
    }

ELOSZOR KIZARO FELTETEL, UTANA SZAZALEKOK
    static int F6_AkciosAr(Aru a)
    {
        if (a.lejarat < DateTime.Today)
            return 0;                             // lejart: 0 Ft
        double ar = a.ar;
        if (a.tipus == "tej")
            ar = ar * 0.75;
        else
            ar = ar * 0.70;
        if (a.keszlet >= 60)
            ar = ar * 0.94;                       // tovabbi -6%
        return (int)Math.Round(ar);
    }

TABLAZATBOL ERTEK + HATVANY (amortizacio evente)
    static int F8_BecsultAr(Aru a)
    {
        double amort;
        switch (a.allapot)
        {
            case "ujszeru": amort = 0.11; break;
            case "megkimelt": amort = 0.20; break;
            case "serult": amort = 0.35; break;
            default: amort = 0.42; break;
        }
        int felar = 0;
        if (a.klima == "manualis") felar = 120000;
        else if (a.klima == "digitalis") felar = 245000;
        else if (a.klima == "digitalis-tobbzonas") felar = 320000;
        int kor = DateTime.Today.Year - a.evjarat;
        double ar = a.ar * Math.Pow(1 - amort, kor) + felar;
        return (int)Math.Round(ar);
    }
